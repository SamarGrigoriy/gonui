# Бюджеты размера релизных бинарников (docs/size.md). Превышение роняет make size.
# Менять только отдельным коммитом с объяснением, на что ушли байты.
# Бюджет — замер + 10 %: тесный, чтобы запас не «съели» незаметно.
# пример             тулчейн  байт      # почему столько
examples/hello       tinygo   26048     # фаза 1: 23 680 после strip + 10 %; цель goals.md — ≤ 65 536
examples/hello       gc       1431488   # фаза 1: 1 301 352 + 10 % (оценка по S1 была 1 625 000)
examples/form        tinygo   45988     # фаза 2: 41 808 после strip + 10 % (оценка size.md была ≤ 160 КБ)
examples/form        gc       1536066   # фаза 2: 1 396 424 + 10 %
examples/chat        tinygo   46270     # 42 064 + 10 %: к 36 848 чата — кнопка «Скопировать ответ» (буфер обмена), многострочный ввод (TextView.OnEnter, подсказка через pango-cairo)
examples/chat        gc       1509033   # 1 371 848 + 10 % (чат с буфером обмена и многострочным вводом)
examples/viewer      tinygo   59576     # просмотрщик картинок: 54 160 после strip + 10 % (os нет: пути из nui.Args)
examples/viewer      gc       1855612   # 1 686 920 + 10 %
